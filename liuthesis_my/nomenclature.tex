\addcontentsline{toc}{chapter}{Nomenclature}

\noindent This section lists the main symbols and abbreviations used in the thesis.

\section*{Symbols}
\begin{description}
\item[$\mathcal{G}=(\mathcal{V},\mathcal{E})$] Directed biological graph, where $\mathcal{V}$ is the set of nodes and $\mathcal{E}$ is the set of edges.
\item[$A$] Adjacency matrix of the biological graph.
\item[$B$] Mini batch size.
\item[$G$] Number of genes used in the model. In the main setting, $G=978$.
\item[$x_{\mathrm{ctl}}$] Matched control profile.
\item[$x_{\mathrm{trt}}$] Treated expression profile.
\item[$x_{\mathrm{drug}}$] Drug target input vector.
\item[$y$] True response vector, defined as $x_{\mathrm{trt}} - x_{\mathrm{ctl}}$.
\item[$\hat{y}$] Predicted response vector.
\item[$c_{\mathrm{ctx}}$] Sample specific cell context vector derived from the matched control profile.
\item[$C_{\mathrm{ctx}}$] Matrix formed by repeating $c_{\mathrm{ctx}}$ across all gene nodes.
\item[$H^{(0)}$] Initial node representation matrix.
\item[$h_{v}^{(L)}$] Final hidden representation of node $v$ after the last graph layer.
\item[$\mu$] Predicted mean response.
\item[$\sigma^{2}$] Predicted variance.
\item[$\log \sigma^{2}$] Predicted log variance.
\item[$\mathcal{L}_{\mathrm{nll}}$] Gaussian negative log likelihood term.
\item[$\mathcal{L}_{\mathrm{pcc}}$] Pearson correlation loss term.
\item[$\mathcal{L}_{\mathrm{full}}$] Main prediction objective, combining NLL and PCC terms.
\item[$\mathcal{L}_{\mathrm{cf}}$] Prediction objective computed from the counterfactual branch.
\item[$\mathcal{L}_{\mathrm{hinge}}$] Counterfactual hinge penalty.
\item[$\lambda_{\mathrm{pcc}}$] Weight of the PCC loss term.
\item[$\lambda_{\mathrm{cf}}$] Weight of the counterfactual loss term.
\item[$\delta$] Margin used in the counterfactual hinge penalty.
\end{description}

\section*{Abbreviations}
\begin{description}
\item[Adam] Adaptive moment estimation optimizer used for training.
\item[CF] Counterfactual.
\item[CF UPert] UPert variant trained with the counterfactual drug constraint.
\item[CellId] Variant that uses a cell identity embedding instead of cell context.
\item[DeepCOP] Deep learning based baseline model used for perturbation response prediction.
\item[DMSO] Dimethyl sulfoxide, used as the solvent control in the experiments.
\item[FFN] Feed forward network.
\item[GAT] Graph attention network.
\item[GNN] Graph neural network.
\item[GSNN] Graph Structured Neural Network.
\item[LINCS] Library of Integrated Network-Based Cellular Signatures.
\item[L1000] Landmark gene expression assay used in LINCS.
\item[MLP] Multilayer perceptron.
\item[MSE] Mean squared error.
\item[NLL] Negative log likelihood.
\item[Neg P@20] Negative precision at 20, used for the strongest down regulated genes.
\item[no-CF UPert] UPert variant trained without the counterfactual drug constraint.
\item[OmniPath] Literature curated database of molecular interactions.
\item[PCC] Pearson correlation coefficient.
\item[Pos P@20] Positive precision at 20, used for the strongest up regulated genes.
\item[ReLU] Rectified linear unit activation function.
\item[UPert] Uncertainty Perturbation, the proposed model in this thesis.
% \item[\texttt{mean\_cell}] Cell specific average baseline computed from the training set.
% \item[\texttt{mean\_drug}] Drug specific average baseline computed from the training set.
% \item[\texttt{mean\_global}] Global average baseline computed from the full training set.
\end{description}
